# Long-form ASR project — pinned to the versions that actually work on
# Windows + RTX 4060 (8GB) + CUDA 12.4 in the `TTS` conda env.
#
# DO NOT pip-install this blindly into a fresh env on Windows — torch must be
# installed FIRST from the official CUDA index, otherwise pip will pull the
# CPU build. Recommended bootstrap:
#
#     conda create -n TTS python=3.10 -y
#     conda activate TTS
#     pip install torch==2.5.1 torchaudio==2.5.1 torchvision==0.20.1 \
#         --index-url https://download.pytorch.org/whl/cu124
#     pip install -r requirements.txt
#     pip install whisperx==3.4.2 --no-deps          # avoid pulling a newer torch stack
#
# Current README and scripts target the local Windows workflow.  Keep the
# faster-whisper / ctranslate2 packages installed because WhisperX imports may
# require them, but the local file ASR backend is not faster-whisper.
